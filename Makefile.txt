# Math 3.0 Absolute Verification Pipeline Automation

PYTHON ?= python
LEAN ?= lean
LEAN_JOBS ?= 1

.PHONY: verify clean environment-audit audit lean-audit regression blindspot gate seal verify-ledger hooks

verify: environment-audit
	@echo "Executing Math 3.0 Universal Gate Suite..."
	$(PYTHON) tools/suite_check.py

audit: environment-audit
	@echo "Executing full Math 3.0 audit stack..."
	$(PYTHON) tools/lean_surface_audit.py --jobs $(LEAN_JOBS) --timeout 180
	$(PYTHON) tools/blindspot_audit.py
	$(PYTHON) tools/audit_hardcoded_nodes.py
	$(PYTHON) tools/meta_blindspot_hunter.py
	$(PYTHON) tools/omega_audit/gate_strict.py check lean/spec_v4 Math30Kernel Math30Interfaces

lean-audit:
	$(PYTHON) tools/lean_surface_audit.py --jobs $(LEAN_JOBS) --timeout 180

regression:
	$(PYTHON) tools/run_regression_tests.py

blindspot:
	$(PYTHON) tools/blindspot_audit.py

gate:
	$(PYTHON) tools/omega_audit/gate_strict.py check lean/spec_v4 Math30Kernel Math30Interfaces

seal:
	$(PYTHON) mcp/t2n/core/repokeeper.py

verify-ledger:
	$(PYTHON) mcp/t2n/core/repokeeper.py --verify
	$(PYTHON) tools/omega_audit/manifest_verify.py

hooks:
	$(PYTHON) tools/omega_audit/install_hooks.py

environment-audit:
	@echo "Checking system toolchain specifications..."
	@$(PYTHON) -c "import flint" || (echo "FATAL: python-flint package missing." && exit 2)
	@$(PYTHON) -c "import z3" || (echo "FATAL: z3-solver Python package missing." && exit 2)
	@$(LEAN) --version || (echo "FATAL: Lean 4 compiler missing from PATH." && exit 2)

clean:
	@echo "Purging temporary run caches from tracking zones..."
	find . -type d -name "__pycache__" -exec rm -rf {} +
	rm -rf mcp/out/*
